import markdown from pathlib import Path import sys

extensions = [ 'fenced_code', # Enables code blocks 'codehilite', # Syntax highlighting in those blocks 'tables' # Markdown table support ]

for md_file in sys.argv: with open(md_file, 'r', encoding='utf-8') as f: content = f.read()

html_content = markdown.markdown(content, extensions=extensions)

full_html = f"""<!DOCTYPE html>

output_file = Path(output_dir) / f"{md_file}.html"

    f.write(full_html)
